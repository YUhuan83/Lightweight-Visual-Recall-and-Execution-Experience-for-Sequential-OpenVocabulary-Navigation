cmake_minimum_required(VERSION 3.8)
project(wildos_ackermann_rpp)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(nav2_regulated_pure_pursuit_controller 1.1.20 EXACT REQUIRED)
foreach(dep nav2_regulated_pure_pursuit_controller nav2_core nav2_costmap_2d nav2_util rclcpp rclcpp_lifecycle pluginlib geometry_msgs nav_msgs tf2)
  find_package(${dep} REQUIRED)
endforeach()
add_library(${PROJECT_NAME} SHARED src/controller.cpp src/progress_goal_checker.cpp)
target_include_directories(${PROJECT_NAME} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME} nav2_regulated_pure_pursuit_controller nav2_core nav2_costmap_2d nav2_util rclcpp rclcpp_lifecycle pluginlib geometry_msgs nav_msgs tf2)
pluginlib_export_plugin_description_file(nav2_core plugin.xml)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_curvature test/test_curvature.cpp)
  target_include_directories(test_curvature PRIVATE include)
  ament_add_gtest(test_controller test/test_controller.cpp)
  ament_target_dependencies(test_controller nav2_regulated_pure_pursuit_controller nav2_core nav2_costmap_2d nav2_util rclcpp rclcpp_lifecycle pluginlib geometry_msgs nav_msgs tf2)
endif()
ament_export_include_directories(include)
ament_export_libraries(${PROJECT_NAME})
ament_package()
