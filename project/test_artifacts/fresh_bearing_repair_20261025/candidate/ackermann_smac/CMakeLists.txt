cmake_minimum_required(VERSION 3.16)
project(wildos_ackermann_smac)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(nav2_smac_planner 1.1.20 EXACT REQUIRED)
find_package(ompl REQUIRED)
foreach(dep nav2_core nav2_costmap_2d nav2_util rclcpp rclcpp_lifecycle pluginlib geometry_msgs nav_msgs tf2 tf2_geometry_msgs)
  find_package(${dep} REQUIRED)
endforeach()
add_library(${PROJECT_NAME} SHARED src/connector.cpp src/planner.cpp src/path_contract.cpp)
target_include_directories(${PROJECT_NAME} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> ${OMPL_INCLUDE_DIRS})
target_link_libraries(${PROJECT_NAME} ${OMPL_LIBRARIES})
ament_target_dependencies(${PROJECT_NAME} nav2_smac_planner nav2_core nav2_costmap_2d nav2_util rclcpp rclcpp_lifecycle pluginlib geometry_msgs nav_msgs tf2 tf2_geometry_msgs)
pluginlib_export_plugin_description_file(nav2_core plugin.xml)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_connector test/test_connector.cpp)
  target_link_libraries(test_connector ${PROJECT_NAME})
  ament_target_dependencies(test_connector nav2_costmap_2d)
  ament_add_gtest(test_planner test/test_planner.cpp)
  target_compile_definitions(test_planner PRIVATE FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/test/fixtures")
  target_link_libraries(test_planner ${PROJECT_NAME})
  ament_target_dependencies(test_planner nav2_core nav2_costmap_2d nav2_smac_planner rclcpp rclcpp_lifecycle pluginlib tf2_ros tf2_geometry_msgs)
  ament_add_gtest(test_path_contract test/test_path_contract.cpp)
  target_compile_definitions(test_path_contract PRIVATE FIXTURE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/test/fixtures")
  target_link_libraries(test_path_contract ${PROJECT_NAME})
endif()
ament_export_include_directories(include)
ament_export_libraries(${PROJECT_NAME})
ament_export_dependencies(nav2_smac_planner nav2_costmap_2d)
ament_package()
