cmake_minimum_required(VERSION 3.16)
project(physical_evidence_fixture)
set(CMAKE_CXX_STANDARD 17)
set(MAPPER_PREFIX "" CACHE PATH "Verified isolated Mapper install prefix")
if(NOT EXISTS "${MAPPER_PREFIX}/lib/libgraphnav_mapper_component.so")
  message(FATAL_ERROR "Verified Mapper prefix is required")
endif()
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(grid_map_core REQUIRED)
find_package(grid_map_ros REQUIRED)
find_package(grid_map_msgs REQUIRED)
find_package(graphnav_msgs REQUIRED)
find_package(pose_graph_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(nanoflann REQUIRED)
add_executable(fixture_export fixture_export.cpp)
target_include_directories(fixture_export PRIVATE "${MAPPER_PREFIX}/include")
target_link_libraries(fixture_export "${MAPPER_PREFIX}/lib/libgraphnav_mapper_component.so" Eigen3::Eigen nanoflann::nanoflann)
ament_target_dependencies(fixture_export rclcpp grid_map_core grid_map_ros grid_map_msgs graphnav_msgs pose_graph_msgs nav_msgs visualization_msgs tf2_ros)
