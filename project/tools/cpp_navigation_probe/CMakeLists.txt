cmake_minimum_required(VERSION 3.16)
project(wildos_navigation_snapshot_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(nav2_smac_planner 1.1.20 EXACT REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(nav2_util REQUIRED)
find_package(tf2 REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
add_executable(collision_snapshot_probe collision_snapshot_probe.cpp)
ament_target_dependencies(collision_snapshot_probe
  nav2_smac_planner nav2_costmap_2d rclcpp rclcpp_lifecycle)
enable_testing()
add_test(NAME collision_controls COMMAND collision_snapshot_probe --self-test)
add_executable(smac_planner_snapshot_probe smac_planner_snapshot_probe.cpp)
ament_target_dependencies(smac_planner_snapshot_probe
  nav2_smac_planner nav2_costmap_2d rclcpp rclcpp_lifecycle nav2_util
  tf2 tf2_ros tf2_geometry_msgs)
add_test(NAME inflation_planner_controls COMMAND smac_planner_snapshot_probe --self-test)
find_package(ompl REQUIRED)
add_executable(anchor_connector_probe anchor_connector_probe.cpp)
target_include_directories(anchor_connector_probe PRIVATE ${OMPL_INCLUDE_DIRS})
target_link_libraries(anchor_connector_probe ${OMPL_LIBRARIES})
ament_target_dependencies(anchor_connector_probe nav2_smac_planner nav2_costmap_2d)
add_test(NAME anchor_connector_controls COMMAND anchor_connector_probe --self-test)
